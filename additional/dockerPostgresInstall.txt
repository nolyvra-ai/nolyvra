##SQL Scripts##
-- JOBS
create table if not exists jobs (
  id text primary key,
  title text not null,
  company text,
  job_type text,
  jd_text text not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- CANDIDATES
create table if not exists candidates (
  id text primary key,
  job_id text not null references jobs(id) on delete cascade,
  name text not null,
  linkedin_url text,
  cv_text text,               -- MVP: store text only (PII disclaimer)
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- ANALYSES (store scores + whole json)
create table if not exists analyses (
  id bigserial primary key,
  candidate_id text not null references candidates(id) on delete cascade,
  job_id text not null references jobs(id) on delete cascade,
  analyzed_at timestamptz not null default now(),

  -- quick fields for dashboards
  consistency_score int,
  capability_score int,
  risk_level text,
  timeline_match_percent int,

  -- store full response for UI
  analysis_json jsonb not null,

  created_at timestamptz not null default now()
);
If index error add the following:
create index if not exists idx_candidates_job_id on candidates(job_id);
create index if not exists idx_analyses_candidate_id on analyses(candidate_id);
create index if not exists idx_analyses_job_id on analyses(job_id);
create index if not exists idx_analyses_analyzed_at on analyses(analyzed_at desc);
